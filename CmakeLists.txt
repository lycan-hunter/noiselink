cmake_minimum_required(VERSION 3.20)
project(NoiseLink VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED) 

add_subdirectory(extern/CLI11)

set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(USE_GNUTLS OFF CACHE BOOL "" FORCE)
set(USE_NICE OFF CACHE BOOL "" FORCE)
add_subdirectory(extern/libdatachannel EXCLUDE_FROM_ALL)

add_executable(${PROJECT_NAME} 
    src/main.cpp
)

target_include_directories(${PROJECT_NAME} PUBLIC 
    include
    src
    extern/CLI11/include
    extern/tomlplusplus/include
)

target_compile_definitions(${PROJECT_NAME} PRIVATE TOML_HEADER_ONLY=1)

target_link_libraries(${PROJECT_NAME} PRIVATE 
    datachannel 
    CLI11::CLI11
    Threads::Threads
    OpenSSL::Crypto
    OpenSSL::SSL
)

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
endif()
